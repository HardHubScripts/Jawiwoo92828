local function H(s) return Color3.fromHex(s) end

local function new(class, props)
  local o = Instance.new(class)
  if props then
    local parent = props.Parent
    for k, v in pairs(props) do if k ~= "Parent" then o[k] = v end end
    if parent then o.Parent = parent end
  end
  return o
end

local function corner(p, r)
  return new("UICorner", { CornerRadius = UDim.new(0, r or 8), Parent = p })
end

local function pad(p, a)
  return new("UIPadding", {
    PaddingLeft = UDim.new(0, a), PaddingRight = UDim.new(0, a),
    PaddingTop = UDim.new(0, a), PaddingBottom = UDim.new(0, a), Parent = p,
  })
end

local function listV(p, gap)
  return new("UIListLayout", {
    FillDirection = Enum.FillDirection.Vertical,
    Padding = UDim.new(0, gap or 6),
    SortOrder = Enum.SortOrder.LayoutOrder,
    Parent = p,
  })
end

local function tween(o, t, props, st, dir)
  local tw = TweenService:Create(o, TweenInfo.new(t, st or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out), props)
  tw:Play()
  return tw
end

local function gradSeq(colors)
  local kp = {}
  for i, c in ipairs(colors) do
    table.insert(kp, ColorSequenceKeypoint.new((i - 1) / math.max(#colors - 1, 1), c))
  end
  return ColorSequence.new(kp)
end

-- Tween seguro: ignora se objeto foi destruído
local function tweenSafe(o, t, props, st, dir)
  if not o or not o.Parent then return end
  return tween(o, t, props, st, dir)
end

  })

    BackgroundTransparency = 1,
    Text                   = tostring(defaultVal),
    Font                   = Enum.Font.GothamBold,

    Parent                 = pill,
  })
  local plus = new("TextButton", {
    Size                   = UDim2.new(0, 28, 1, 0),

    TextSize               = 15,

  })
  -- divisores verticais
  new("Frame", { Size = UDim2.fromOffset(1, 18), Position = UDim2.new(0, 28, 0.5, -9),
    BackgroundColor3 = C.Sub, BackgroundTransparency = 0.5, BorderSizePixel = 0, Parent = pill })

    BackgroundColor3 = C.Sub, BackgroundTransparency = 0.5, BorderSizePixel = 0, Parent = pill })

  minus.MouseButton1Click:Connect(function()

-- ÚNICA conexão RenderStepped (vs N bindings do original)
local renderConn
local renderQueue = {}

local function queueRenderUpdate(key, fn)
  renderQueue[key] = fn
  if renderConn then return end
  renderConn = RunService.RenderStepped:Connect(function()
    for k, f in pairs(renderQueue) do
      local ok, err = pcall(f)
      if not ok then renderQueue[k] = nil end
    end
  end)
end

local function unqueueRenderUpdate(key)
  renderQueue[key] = nil
  if next(renderQueue) == nil and renderConn then
    renderConn:Disconnect()
    renderConn = nil
  end
end

local function isWeaponAllowed(name)
  return Whitelist[name] == true
end

local function gettable(filter)
  local results = {}
  if not itemdatabase then return results end
  for key in pairs(itemdatabase) do
    if type(key) == "string" and isWeaponAllowed(key) then
      if filter == "" or key:lower():find(filter:lower(), 1, true) then
        table.insert(results, key)
      end
    end
  end
  table.sort(results)
  return results
end

-- Carrega database
task.spawn(function()
  local ok, err = pcall(function()
    itemdatabase = require(
      ReplicatedStorage:WaitForChild("Database"):WaitForChild("Sync"):WaitForChild("Item")
    )
  end)
  if ok then
    setStatus("Ready — " .. #gettable("") .. " items", C.Good)
  else
    setStatus("DB: " .. tostring(err):sub(1, 40), C.Bad)
  end
end)

-- ADD TO INVENTORY (sem BreakJoints a cada spawn)
-- ============================================================
local function addToInventory(itemId, amount)
  inventory[itemId] = (inventory[itemId] or 0) + (amount or 1)

  local key = "Inv_" .. itemId
  queueRenderUpdate(key, function()
    if PlayerWeapons and PlayerWeapons.Owned then
      PlayerWeapons.Owned[itemId] = inventory[itemId]
    end
  end)

  -- Refresh UI
  local pGui = LP:WaitForChild("PlayerGui")
  local invGui = pGui:FindFirstChild("Inventory") or pGui:FindFirstChild("MainGui")
  if invGui then
    local invFrame = invGui:FindFirstChild("InventoryFrame", true)
    if invFrame then
      invFrame.Visible = false
      task.wait(0.08)
      invFrame.Visible = true
    end
  end
end

local CYCLE = { C.Gold, C.Pink, C.Mint, C.Cyan }

local function injectHighlight(f, boxGui)
  if not f or f:FindFirstChild("__THL") then return end
  new("BoolValue", { Name = "__THL", Parent = f })

  local bg = new("Frame", {
    Name = "__THLBg",
    Size = UDim2.new(1, 0, 1, 0),
    BackgroundColor3 = C.Gold,
    BackgroundTransparency = 0.7,
    BorderSizePixel = 0,
    ZIndex = (f.ZIndex or 1) + 3,
    Parent = f,
  })
  corner(bg, 3)

  local glowHL = new("ImageLabel", {
    Image = "rbxassetid://5028857084",
    Size = UDim2.fromScale(1.4, 1.4),
    AnchorPoint = Vector2.new(0.5, 0.5),
    Position = UDim2.fromScale(0.5, 0.5),
    BackgroundTransparency = 1,
    ImageColor3 = C.Gold,
    ImageTransparency = 0.35,
    ZIndex = (f.ZIndex or 1) + 2,
    Parent = f,
  })

  local hs = new("UIStroke", {
    Thickness = 3,
    Color = C.Gold,
    Transparency = 0,
    ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    Parent = f,
  })

  -- Ciclo de cores (limitado a 20 iterações, depois para)
  local function colorCycle(idx)
    if not hs or not hs.Parent or not boxGui.Parent or idx > 20 then return end
    local c = CYCLE[(idx - 1) % #CYCLE + 1]
    tweenSafe(hs, 0.4, { Color = c })
    tweenSafe(bg, 0.4, { BackgroundColor3 = c })
    tweenSafe(glowHL, 0.4, { ImageColor3 = c })
    task.wait(0.4)
    colorCycle(idx + 1)
  end
  task.spawn(colorCycle, 1)

  -- Pulse (limitado a 15 ciclos)
  local function pulse(idx)
    if not glowHL or not glowHL.Parent or idx > 15 then return end
    tweenSafe(glowHL, 0.4, { ImageTransparency = 0.15 })
    task.wait(0.4)
    tweenSafe(glowHL, 0.4, { ImageTransparency = 0.5 })
    task.wait(0.4)
    pulse(idx + 1)
  end
  task.spawn(pulse, 1)
end

local function addItemHighlight(animDur, targetItem)
  task.spawn(function()
    local pGui = LP:WaitForChild("PlayerGui")
    local target = targetItem:lower()

    -- Espera pela GUI da caixa (com timeout)
    local boxGui
    for _ = 1, 30 do
      boxGui = pGui:FindFirstChild("MysteryBoxOpen")
      if boxGui then break end
      task.wait(0.08)
    end
    if not boxGui then return end

    local container
    pcall(function() container = boxGui:WaitForChild("Container", 2) end)
    if not container then return end

    local function tryHighlight(child)
      if not (child:IsA("GuiObject") and not child:IsA("UILayout") and not child:IsA("UIPadding")) then return end
      for _, desc in ipairs(child:GetDescendants()) do
        if desc:IsA("TextLabel") and desc.Text:lower():find(target, 1, true) then
          injectHighlight(child, boxGui)
          if desc.Parent and desc.Parent ~= child then
            injectHighlight(desc.Parent, boxGui)
          end
          return
        end
      end
    end

    local function processMain(mainFrame)
      local mc
      pcall(function()
        mc = mainFrame:WaitForChild("Container", 2):WaitForChild("Background", 2)
          :WaitForChild("ItemContainer", 2):WaitForChild("OffsetContainer", 2)
          :WaitForChild("MainContainer", 2)
      end)
      if not mc then return end
      for _, child in ipairs(mc:GetChildren()) do tryHighlight(child) end
      local conn
      conn = mc.ChildAdded:Connect(function(child)
        task.defer(function() tryHighlight(child) end)
      end)
      -- Auto-cleanup após a duração da animação
      task.delay(animDur + 2, function()
        if conn then conn:Disconnect() end
      end)
    end

    for _, f in ipairs(container:GetChildren()) do
      if f.Name == "Main" then task.spawn(function() processMain(f) end) end
    end
    local conn2
    conn2 = container.ChildAdded:Connect(function(f)
      if f.Name == "Main" then task.spawn(function() processMain(f) end) end
    end)
    task.delay(animDur + 2, function()
      if conn2 then conn2:Disconnect() end
    end)
  end)
end

-- ============================================================
-- RUN SPAWN (com pcall em tudo)
-- ============================================================
local function runSpawn()
  if spawning or busy then return end
  if itemToSpawn == "" then setStatus("Select an item", C.Bad); return end

  spawning = true
  busy = true
  spawnBtn.Text = "● Spawning"
  setStatus("Unboxing...", C.Mint)

  task.spawn(function()
    local amount = getSpawnAmt()

    -- Monta payload
    local payload = {}
    for _ = 1, amount do
      table.insert(payload, { MysteryBoxId = CRATE, RewardedItemId = itemToSpawn })
    end

    -- Chama o remote com pcall
    local ok, err = pcall(function()
      _BC:Fire(payload)
    end)
    if not ok then
      setStatus("Remote error: " .. tostring(err):sub(1, 30), C.Bad)
      spawning = false
      busy = false
      spawnBtn.Text = "Spawn"
      return
    end

    addToInventory(itemToSpawn, amount)

    -- Progress bar
    local animDur = 4.5
    tweenSafe(progFill, animDur, { Size = UDim2.new(1, 0, 1, 0) }, Enum.EasingStyle.Linear)

    addItemHighlight(animDur, itemToSpawn)

    task.wait(animDur)
    playLandEffect(itemToSpawn, amount)
    task.wait(2)

    tweenSafe(progFill, 0.35, { Size = UDim2.new(0, 0, 1, 0) })
    setStatus("+" .. amount .. "x " .. itemToSpawn, C.Good)
    task.wait(1)
    setStatus("Ready", C.Sub)
    spawnBtn.Text = "Spawn"
    spawning = false
    busy = false
  end)
end

-- DUPE (com pcall)
-- ============================================================
local function runDupe()
  if itemToSpawn == "" then setDupeStatus("Select an item", C.Bad); return end

  local owned = PlayerWeapons and PlayerWeapons.Owned
  local current = (owned and owned[itemToSpawn]) or 0
  local dupeAmount = getDupeAmt()
  local newTotal = current + dupeAmount

  PlayerWeapons.Owned[itemToSpawn] = newTotal
  inventory[itemToSpawn] = newTotal

  local key = "Dupe_" .. itemToSpawn
  queueRenderUpdate(key, function()
    if PlayerWeapons and PlayerWeapons.Owned then
      PlayerWeapons.Owned[itemToSpawn] = inventory[itemToSpawn]
    end
  end)

  pcall(function()
    ReplicatedStorage.Remotes.Inventory.InventoryDataChanged:Fire()
  end)

  setDupeStatus("+" .. dupeAmount .. "  total: " .. newTotal, C.Good)
end

  end)
else
  UserInputService.InputBegan:Connect(function(i, gp)
    if gp then return end
    if i.KeyCode == Enum.KeyCode.RightShift then
      frame.Visible = not frame.Visible
    end
  end)
end
